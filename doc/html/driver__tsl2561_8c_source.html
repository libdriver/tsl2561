148static uint32_t a_tsl2561_calculate_lux(uint16_t package, uint16_t gain, uint16_t t, uint16_t ch0, uint16_t ch1)
149{
150 uint32_t ch_scale;
151 uint32_t channel_1;
152 uint32_t channel_0;
153 uint32_t temp;
154 uint32_t ratio1 = 0;
155 uint32_t ratio;
156 uint32_t lux;
157 uint16_t b,m;
158
159 switch (t) /* choose integration time */
160 {
161 case 0 :
162 {
163 ch_scale = TSL2561_LUX_CHSCALE_TINT0; /* 13.7ms */
164
165 break; /* break */
166 }
167 case 1 :
168 {
169 ch_scale = TSL2561_LUX_CHSCALE_TINT1; /* 101ms */
170
171 break; /* break */
172 }
173 default :
174 {
175 ch_scale = 1 << TSL2561_LUX_CHSCALE; /* assume no scaling */
176
177 break; /* break */
178 }
179 }
180 if (!gain) /* check gain */
181 {
182 ch_scale = ch_scale << 4; /* cale 1X to 16X */
183 }
184 channel_0 = (ch0 * ch_scale) >> TSL2561_LUX_CHSCALE; /* set channel 0 */
185 channel_1 = (ch1 * ch_scale) >> TSL2561_LUX_CHSCALE; /* set channel 1 */
186 if (channel_0 != 0) /* check channel 0 */
187 {
188 ratio1 = (channel_1<<(TSL2561_LUX_RATIOSCALE+1)) / channel_0; /* get ratio */
189 }
190 ratio = (ratio1 + 1) >> 1; /* right shift 1 */
191 if (package == TSL2561_PACKAGE_CS) /* choose CS package */
192 {
193 if (ratio <= TSL2561_LUX_K1C) /* if K1C */
194 {
195 b = TSL2561_LUX_B1C; /* set B1C */
196 m = TSL2561_LUX_M1C; /* set M1C */
197 }
198 else if (ratio <= TSL2561_LUX_K2C) /* if K2C */

274 m = TSL2561_LUX_M8T; /* set M8T */
275 }

284 lux = temp >> TSL2561_LUX_LUXSCALE; /* calculate lux */
285
